# zenix -- a Sonoma-flavoured GRUB theme.
#
# Modelled on the macOS Startup Manager: flat near-black ground, one large
# disk icon per entry, generous negative space, and a selection that is a soft
# fill rather than a button. Deliberately no title, no keyboard hints and no
# colour -- those read as a Linux theme.
#
# Font names must match the NAME baked into the .pf2 files exactly; grub-mkfont
# composes them as "<family> <style> <size>". See build.sh.

title-text: ""

desktop-image: "background.png"
desktop-color: "#181926"

terminal-font: "Inter Regular 16"
terminal-left: "8%"
terminal-top: "10%"
terminal-width: "84%"
terminal-height: "78%"
terminal-border: "0"

message-font: "Inter Regular 16"
message-color: "#cad3f5"
message-bg-color: "#181926"

# ---- the entries ----

+ boot_menu {
    left   = 50%-290
    top    = 26%
    width  = 580
    height = 46%

    item_font           = "Inter Regular 20"
    item_color          = "#a5adcb"
    selected_item_color = "#cad3f5"

    # tall rows so each icon has room to breathe, the way the Startup
    # Manager spaces its disks
    item_height  = 124
    item_spacing = 6
    item_padding = 18

    icon_width      = 96
    icon_height     = 96
    item_icon_space = 24

    selected_item_pixmap_style = "select_*.png"
}

# ---- countdown ----
#
# Thin, white, and low on the screen -- the bar that sits under the Apple logo
# while a Mac boots.

+ progress_bar {
    id           = "__timeout__"
    left         = 50%-120
    top          = 78%
    width        = 240
    height       = 3
    text         = ""
    bg_color     = "#363a4f"
    fg_color     = "#cad3f5"
    border_color = "#363a4f"
}
